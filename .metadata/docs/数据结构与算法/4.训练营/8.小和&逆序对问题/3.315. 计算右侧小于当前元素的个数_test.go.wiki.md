// 右侧小于当前元素的个数
func merge1(arr []*Node, L, M, R int, ans []int) {
    help := make([]*Node, R-L+1)
    i := L
    j := M+1
    index := 0
    for i <= M && j <= R {
        if arr[i].Value <= arr[j].Value {
            ans[arr[i].Index] += (j-1) - (M+1) + 1
            help[index] = arr[i]
            i++
            index++
        }else {
            help[index] = arr[j]
            j++
            index++
        }
    }

    for i <= M {
        ans[arr[i].Index] += R - (M+1) + 1
        help[index] = arr[i]
        i++
        index++
    }

    for j <= R {
        help[index] = arr[j]
        j++
        index++
    }
    copy(arr[L: R+1], help)
}

// 右侧大于当前元素的个数（小和问题）
func merge2(arr []*Node, L, M, R int, ans []int) {
    help := make([]*Node, R-L+1)
	i := L
	j := M + 1
	index := 0
	for i <= M && j <= R {
		if arr[i].Value < arr[j].Value {
			ans[arr[i].Index] += R - j + 1
			help[index] = arr[i]
			i++
			index++
		} else {
			help[index] = arr[j]
			j++
			index++
		}
	}

	for i <= M {
		help[index] = arr[i]
		i++
		index++
	}

	for j <= R {
		help[index] = arr[j]
		j++
		index++
	}
	copy(arr[L:R+1], help)
}
