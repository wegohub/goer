import "fmt"

// 如果第一个玩家获胜，由于第一个玩家是先手，则次数游戏板中 "X" 的数量比 "O" 的数量多 1。
// 如果第二个玩家获胜，则 "X" 的数量与 "O" 的数量相同。
func validTicTacToe(board []string) bool {
	XCount := 0
	OCount := 0
	for i := 0; i < 3; i++ {
		for j := 0; j < 3; j++ {
			if board[i][j] == 'X' {
				XCount++
			}
			if board[i][j] == 'O' {
				OCount++
			}
		}
	}
	if XCount == 0 && OCount == 0 {
		return true
	}

	win := whoWin(board)
    if len(win) > 1 {
        return false
    }
    fmt.Println(string(win))
	if len(win) == 0 { // 没有人赢
		return XCount == OCount || XCount == OCount+1
	} else if win[0] == 'X' {
		return XCount == OCount+1
	} else {
		return OCount == XCount
	}
}

func whoWin(board []string) []byte {
    ans := make([]byte, 0)
	// 行有没有相同的
	for i := 0; i < 3; i++ {
		if board[i][0] != ' ' && board[i][0] == board[i][1] && board[i][1] == board[i][2] {
            if len(ans) == 0 || ans[0] != board[i][0] {
                ans = append(ans, board[i][0]) 
            }
			
		}
	}

	// 列有没有相同的
	for j := 0; j < 3; j++ {
		if board[0][j] != ' ' && board[0][j] == board[1][j] && board[1][j] == board[2][j] {
            if len(ans) == 0 || ans[0] != board[0][j]{
                ans = append(ans, board[0][j]) 
            }
		}
	}

	// 左对角线
	if board[0][0] != ' ' && board[0][0] == board[1][1] && board[1][1] == board[2][2] {
        if len(ans) == 0 || ans[0] != board[0][0] {
            ans = append(ans, board[0][0]) 
        }
		
	}

	// 右对角线
	if board[0][2] != ' ' && board[0][2] == board[1][1] && board[1][1] == board[2][0] {
        if len(ans) == 0 || ans[0] != board[0][2] {
            ans = append(ans, board[0][2]) 
        }
		
	}

	return ans
}
