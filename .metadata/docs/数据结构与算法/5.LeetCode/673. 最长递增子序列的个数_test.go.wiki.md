```
func findLIS(nums []int) [][]int {
	if len(nums) == 0 {
		return [][]int{}
	}

	// tails[i] represents the smallest tail of all increasing subsequences of length i+1
	tails := make([]int, len(nums))
	// prev[i] represents the index of the previous number in the LIS ending with nums[i]
	prev := make([]int, len(nums))
	// Length of the longest increasing subsequence found so far
	length := 0

	for i, num := range nums {
		// Binary search to find the position to update
		left, right := 0, length
		for left < right {
			mid := (left + right) / 2
			if tails[mid] < num {
				left = mid + 1
			} else {
				right = mid
			}
		}

		// Update the tails array
		tails[left] = num
		prev[i] = left
		if left == length {
			length++
		}
	}

	// Reconstruct all LIS
	lisList := [][]int{}
	stack := []int{}
	findAllLIS(nums, prev, length-1, len(nums)-1, &stack, &lisList)

	return lisList
}

// Recursive function to find all LIS
func findAllLIS(nums, prev []int, targetLen, currentIndex int, stack *[]int, lisList *[][]int) {
	if targetLen == -1 {
		lis := make([]int, len(*stack))
		copy(lis, *stack)
		*lisList = append(*lisList, lis)
		return
	}

	if currentIndex < 0 {
		return
	}

	// Include the current element in the LIS
	o := *stack
	if prev[currentIndex] == targetLen && (len(o) == 0 || nums[currentIndex] != o[0]) {
		*stack = append([]int{nums[currentIndex]}, *stack...)
		findAllLIS(nums, prev, targetLen-1, currentIndex-1, stack, lisList)
		*stack = (*stack)[1:]
	}

	// Exclude the current element from the LIS
	findAllLIS(nums, prev, targetLen, currentIndex-1, stack, lisList)
}
```
