```
func largestRectangleArea(heights []int) int {
	if len(heights) == 0 {
		return 0
	}
	maxArea := 0

	// 准备一个单调栈，存下标
	stack := list.New()
	for i := 0; i < len(heights); i++ {
		// 结算答案
		for stack.Len() > 0 && heights[stack.Front().Value.(int)] >= heights[i] {
			cur := stack.Front().Value.(int)
			stack.Remove(stack.Front())
			// 右边离它最近的最小的是i
			right := i
			// 左边离它最近的最小是它盖着的数，没有就是-1
			left := -1
			for stack.Len() > 0 {
				left = stack.Front().Value.(int)
			}
			// ((right - 1) - (left + 1)) + 1  当前弹出的数高度是最小的，其他位置都比它高，否则cur都被结算了
			curArea := (right - left - 1) * heights[cur]
			maxArea = int(math.Max(float64(curArea), float64(maxArea)))
		}
		stack.PushFront(i)
	}

	// 栈未空， 右边离它小的就没有
	for stack.Len() > 0 {
		cur := stack.Front().Value.(int)
		stack.Remove(stack.Front())
		// 右边离它最近的最小的是数组的长度
		right := len(heights)
		// 左边离它最近的最小是它盖着的数，没有就是-1
		left := -1
		if stack.Len() > 0 {
			left = stack.Front().Value.(int)
		}
		curArea := (right - left - 1) * heights[cur]
		maxArea = int(math.Max(float64(curArea), float64(maxArea)))
	}

	return maxArea
}
```
