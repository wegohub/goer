## 总结

- 功能：大文章在命中了哪些串

- 以e结尾后缀指向以e结尾的最长前缀，就是fail指针的指向(每次都匹配最长，所以是贪婪模式)
  
  ![image.png](../../../../static/upload/d311bff2-6b84-4904-94ee-2408b7937912.png)

你提到的没错，使用读写锁（`sync.RWMutex`）虽然可以保证并发安全，但在高并发场景下，如果词库很大，更新词库确实会非常耗时，因为写锁会阻塞所有的读操作。为了优化这种情况，可以考虑以下几种策略：

#### 1. 分段锁
将AC自动机分成多个段，每个段使用独立的读写锁。这样可以减少更新词库时的锁冲突，提高并发性能。

#### 2. 双缓冲技术
使用双缓冲技术，即维护两个AC自动机实例，一个用于读操作，一个用于写操作。当需要更新词库时，在后台构建新的AC自动机实例，构建完成后切换指针，使读操作切换到新的实例。

#### 3. 异步更新
将词库更新操作放到后台异步进行，前端读操作继续使用旧的AC自动机实例。更新完成后，通过某种机制（如通道或回调）通知前端切换到新的实例。

#### 4. 增量更新
只更新发生变化的部分，而不是整个AC自动机。例如，只更新新增的模式串，而不重新构建整个AC自动机。

以下是一个使用双缓冲技术的示例：

```go
package main

import (
    "fmt"
    "sync"
    "sync/atomic"
)

type ACNode struct {
    children map[rune]*ACNode
    fail     *ACNode
    isEnd    bool
    length   int
}

func NewACNode() *ACNode {
    return &ACNode{
        children: make(map[rune]*ACNode),
        fail:     nil,
        isEnd:    false,
        length:   0,
    }
}

type ACAutomaton struct {
    instances [2]*ACNode
    current   int64
    mu        sync.RWMutex
}

func NewACAutomaton() *ACAutomaton {
    return &ACAutomaton{
        instances: [2]*ACNode{NewACNode(), NewACNode()},
        current:   0,
    }
}

func (ac *ACAutomaton) getCurrent() *ACNode {
    return ac.instances[atomic.LoadInt64(&ac.current)]
}

func (ac *ACAutomaton) getNext() *ACNode {
    return ac.instances[1-atomic.LoadInt64(&ac.current)]
}

func (ac *ACAutomaton) switchInstance() {
    atomic.StoreInt64(&ac.current, 1-atomic.LoadInt64(&ac.current))
}

func (ac *ACAutomaton) Insert(word string) {
    ac.mu.Lock()
    defer ac.mu.Unlock()

    node := ac.getNext()
    for _, ch := range word {
        if _, exists := node.children[ch]; !exists {
            node.children[ch] = NewACNode()
        }
        node = node.children[ch]
    }
    node.isEnd = true
    node.length = len(word)
}

func (ac *ACAutomaton) BuildFailPointer() {
    ac.mu.Lock()
    defer ac.mu.Unlock()

    root := ac.getNext()
    queue := []*ACNode{}
    root.fail = nil
    queue = append(queue, root)

    for len(queue) > 0 {
        current := queue[0]
        queue = queue[1:]

        for ch, child := range current.children {
            if current == root {
                child.fail = root
            } else {
                fail := current.fail
                for fail != nil {
                    if next, exists := fail.children[ch]; exists {
                        child.fail = next
                        break
                    }
                    fail = fail.fail
                }
                if fail == nil {
                    child.fail = root
                }
            }
            queue = append(queue, child)
        }
    }
}

func (ac *ACAutomaton) Search(text string) map[string]bool {
    ac.mu.RLock()
    defer ac.mu.RUnlock()

    result := make(map[string]bool)
    node := ac.getCurrent()

    for i, ch := range text {
        for node != node.fail && node.children[ch] == nil {
            node = node.fail
        }
        if next, exists := node.children[ch]; exists {
            node = next
        }
        if node.isEnd {
            result[text[i-node.length+1:i+1]] = true
        }
    }

    return result
}

func (ac *ACAutomaton) Update() {
    ac.mu.Lock()
    defer ac.mu.Unlock()

    ac.BuildFailPointer()
    ac.switchInstance()
}

func main() {
    ac := NewACAutomaton()
    ac.Insert("he")
    ac.Insert("she")
    ac.Insert("his")
    ac.Insert("hers")

    ac.Update()

    text := "ahishers"
    matches := ac.Search(text)

    for match := range matches {
        fmt.Println("Matched:", match)
    }
}
```

在这个示例中，我们使用了双缓冲技术来减少更新词库时的锁冲突。通过维护两个AC自动机实例，一个用于读操作，一个用于写操作，可以在更新词库时避免阻塞读操作，从而提高并发性能。
