- 接口是由使用方定义的

- 面向对象设计 -> interface就近申明 -> mock打桩 -> 单测逻辑

- 基本使用

  - 定义接口, 申明mock代码生成命令

```
//go:generate mockgen -source=person.go -destination=mock/mock_person.go -package=mock
type Person interface {
	Speak() string
	Walk(args string) string
}
```

  - 编写单测

```
func Test_mock_person(t *testing.T) {
	// 初始化gomock控制器
	ctrl := gomock.NewController(t)
	defer ctrl.Finish()

	// 实例化接口实现
	mockPerson := mock.NewMockPerson(ctrl)

	// 单次调用
	mockPerson.EXPECT().Speak().DoAndReturn(func() string {
		return "chinese"
	})

	resp := mockPerson.Speak()
	t.Log("单次调用: ", resp)

	// 多次调用, 也可以把单次的代码放for循环
	testTimes := 3
	mockPerson.EXPECT().Speak().Return("english").Times(testTimes)
	for i := 0; i < testTimes; i++ {
		t.Log("多次调用: ", mockPerson.Speak())
	}

	// 任意次
	mockPerson.EXPECT().Speak().Return("any").AnyTimes()
	t.Log("任意次", mockPerson.Speak())

	// 传入实参, 这里Walk的参数是期望的参数，调用时不一致会导致测试不通过
	mockPerson.EXPECT().Walk("实参").DoAndReturn(func(args string) string {
		return args + "-闭包处理后"
	})
	t.Log(mockPerson.Walk("实参"))
}
```

- gomonkey 更加灵活，不需要基于接口打桩，应用于某些粗暴的场景

  - 场景：当前时间也是不稳定的，如果按照gomock的方式就需要将`now := time.Now()`这行代码封装然后打桩就太重了，就可以使用gomonkey这种方式直接替换运行时now的内存地址
```
func dosomething() {
  now := time.Now()
}

type Boy struct {
}

func (obj *Boy) Speak() string {
	return "chinese"
}

func Test_monkey111(t *testing.T) {
	boy := &Boy{} // 使用指针类型实例

	// 使用 gomonkey 打桩 Speak 方法
	patches := gomonkey.ApplyMethod(boy, "Speak", func(obj *Boy) string {
		return "chinese"
	})
	defer patches.Reset()

	// 调用 Speak 方法并验证结果
	result := boy.Speak()
	t.Log(result)
}

func Laugh() string {
	return "english"
}

func Test_monkey2(t *testing.T) {
	patch := gomonkey.ApplyFunc(Laugh, func() string {
		return "55555"
	})
	defer patch.Reset()
	t.Log(Laugh())
}

```
