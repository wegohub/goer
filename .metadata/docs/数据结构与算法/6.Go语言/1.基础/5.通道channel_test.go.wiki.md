**- 总结**
  - **从关闭的通道中读取数据不会阻塞，返回类型的零值，往关闭的通道中写入数据产生panic**
  - **通道被重复关闭会产生panic**
  - **关闭一个未初始化的通道会panic**
  - **关闭通道会唤醒所有的读写等待队列中的g, 读队列会正常返回，写队列中的g会产生panic**
  - 对于未初始化的 chan，写入操作会引发死锁
    ```
    // 让协程陷入阻塞
    gopark(nil, nil, waitReasonChanSendNilChan, traceEvGoStop, 2)
    // 唤醒协程
    goready(gp, 3)
    ```
  - 对于已关闭的 chan，写入操作会引发 panic
  - 读空 channel，park 挂起，引起死锁
  - channel 已关闭且内部无元素，不会阻塞返回类型零值，上层通过 v,ok := <-c的ok判断是有其他协程放入了零值还是因为通道被关闭返回的零值。

```
type Wait struct {
	Key   int
	Value int
}

func NewMyConCurrentMap() MyConCurrentMap {
	return MyConCurrentMap{
		mp: sync.Map{},
		c:  make(chan *Wait),
		n:  0,
	}
}

type MyConCurrentMap struct {
	mp sync.Map   // 并发map, 可以用读写锁保护
	c  chan *Wait // 无缓存通道
	n  int32      // 等待协程的数量
}

func (m *MyConCurrentMap) Put(k, v int) {
	if _, ok := m.mp.Load(k); ok { // 更新操作
		m.mp.Store(k, v)
	} else { // 新增操作
		m.mp.Store(k, v)
		for i := 0; i < int(m.n); i++ {
			m.c <- &Wait{Key: k, Value: v}
		}
	}
}

func (m *MyConCurrentMap) Get(k int, maxWaitingDuration time.Duration) (int, error) {
	val, ok := m.mp.Load(k)
	if ok {
		return val.(int), nil
	}

	atomic.AddInt32(&m.n, 1)
	ctx, cancel := context.WithTimeout(context.TODO(), maxWaitingDuration)
	defer func() {
		cancel()
		atomic.AddInt32(&m.n, -1)
	}()
	for {
		select {
		case <-ctx.Done():
			return 0, ctx.Err()
		case cur := <-m.c:
			if cur.Key == k {
				return cur.Value, nil
			}
		}
	}
}

func Test_NewMyConCurrentMap(t *testing.T) {
	mp := NewMyConCurrentMap()

	testTimes := 10000

	go func() {
		for i := 0; i < testTimes; i++ {
			time.Sleep(time.Microsecond * 10)
			mp.Put(i, i)
		}
	}()

	wg := sync.WaitGroup{}
	for i := 0; i < testTimes; i++ {
		wg.Add(1)
		go func(idx int) {
			defer wg.Done()
			v, err := mp.Get(idx, time.Second*3)
			if err != nil {
				fmt.Println(err)
				return
			}
			fmt.Println(v)
		}(i)
	}
	wg.Wait()
}
```

```
func NewMyChan() MyChan {
	return MyChan{
		ch: make(chan struct{}),
	}
}

type MyChan struct {
	sync.Once
	ch chan struct{}
}

func (obj *MyChan) Close() {
	obj.Do(func() {
		close(obj.ch)
	})
}

func NewMyConCurrentMap2() MyConCurrentMap2 {
	return MyConCurrentMap2{
		mp:  make(map[int]int),
		key: make(map[int]chan struct{}),
	}
}

type MyConCurrentMap2 struct {
	sync.Mutex                       // 写锁保护map
	mp         map[int]int           // 数据map
	key        map[int]chan struct{} // key对于的阻塞通道
}

func (m *MyConCurrentMap2) Put(k, v int) {
	m.Lock()
	defer m.Unlock()
	// 修改操作
	if _, ok := m.mp[k]; ok {
		m.mp[k] = v
		return
	}
	// 新增操作
	m.mp[k] = v

	// 唤醒阻塞的读协程
	if c, ok := m.key[k]; ok {
		// close(c)
		// c 有可能被重复关闭
		// 解决办法：
		// 1. select 多路复用
		// 2. 封装chan, 配合sync.Once保证close只发生一次
		// 3. 删除m.key[k]
		select {
		case <-c: // 能进这个case说明chan已经被关闭了
			return
		default: // 否则关闭通道
			close(c)
		}

	}
}

func (m *MyConCurrentMap2) Get(k int, maxWaitingDuration time.Duration) (int, error) {
	m.Lock()
	if v, ok := m.mp[k]; ok {
		m.Unlock()
		return v, nil
	}

	// 阻塞等待
	c, ok := m.key[k]
	if !ok {
		c = make(chan struct{})
		m.key[k] = c
	}

	// 这里必须先解锁，后续流程会挂起，这里不解锁就死锁了
	m.Unlock()

	select {
	case <-time.After(maxWaitingDuration):
		return -1, errors.New("timeout")
	case <-c:
		// 这里还要加锁
		m.Lock()
		v := m.mp[k]
		m.Unlock()
		return v, nil
	}
}

func Test_NewMyConCurrentMap2(t *testing.T) {
	mp := NewMyConCurrentMap2()

	testTimes := 10000

	go func() {
		for i := 0; i < testTimes; i++ {
			time.Sleep(time.Microsecond * 10)
			mp.Put(i, i)
		}
	}()

	wg := sync.WaitGroup{}
	for i := 0; i < testTimes; i++ {
		wg.Add(1)
		go func(idx int) {
			defer wg.Done()
			v, err := mp.Get(idx, time.Second*3)
			if err != nil {
				fmt.Println(err)
				return
			}
			fmt.Println(v)
		}(i)
	}
	wg.Wait()
}
```

```
type hchan struct {
    qcount   uint           // total data in the queue
    dataqsiz uint           // size of the circular queue
    buf      unsafe.Pointer // points to an array of dataqsiz elements
    elemsize uint16
    closed   uint32
    elemtype *_type // element type
    sendx    uint   // send index
    recvx    uint   // receive index
    recvq    waitq  // list of recv waiters
    sendq    waitq  // list of send waiters
    
    lock mutex
}
hchan：channel 数据结构
• qcount：当前 channel 中存在多少个元素；
• dataqsize: 当前 channel 能存放的元素容量；
• buf：channel 中用于存放元素的环形缓冲区；
• elemsize：channel 元素类型的大小；
• closed：标识 channel 是否关闭；
• elemtype：channel 元素类型；
• sendx：发送元素进入环形缓冲区的 index；
• recvx：接收元素所处的环形缓冲区的 index；
• recvq：因接收而陷入阻塞的协程队列；
• sendq：因发送而陷入阻塞的协程队列；

1.2 waitq
type waitq struct {
    first *sudog
    last  *sudog
}
waitq：阻塞的协程队列
• first：队列头部
• last：队列尾部

1.3 sudog
type sudog struct {
    g *g
    next *sudog
    prev *sudog
    elem unsafe.Pointer // data element (may point to stack)

    isSelect bool

    c        *hchan 
}
sudog：用于包装协程的节点
• g：goroutine，协程；
• next：队列中的下一个节点；
• prev：队列中的前一个节点；
• elem: 读取/写入 channel 的数据的容器;
• isSelect：标识当前协程是否处在 select 多路复用的流程中；
• c：标识与当前 sudog 交互的 chan.
```
