```go
    viper.AddConfigPath("./configs")

    viper.SetDefault("server.mode", "debug")
    viper.SetDefault("database.host", "localhost")

        return nil, err

        return nil, err
    }
    
    return &config, nil
```

```go
    }
    
    return db, nil
```

**internal/repository/user_repository.go**

```go
package repository

import (
    "your-project/internal/model"
    "gorm.io/gorm"
)

type UserRepository interface {
    Create(user *model.User) error
    GetByID(id uint) (*model.User, error)
    GetByEmail(email string) (*model.User, error)
    Update(user *model.User) error
    Delete(id uint) error
    List(offset, limit int) ([]*model.User, error)
}

type userRepository struct {
    db *gorm.DB
}

func NewUserRepository(db *gorm.DB) UserRepository {
    return &userRepository{db: db}
}

func (r *userRepository) Create(user *model.User) error {
    return r.db.Create(user).Error
}

func (r *userRepository) GetByID(id uint) (*model.User, error) {
    var user model.User
    err := r.db.First(&user, id).Error
    if err != nil {
        return nil, err
    }
    return &user, nil
}

func (r *userRepository) GetByEmail(email string) (*model.User, error) {
    var user model.User
    err := r.db.Where("email = ?", email).First(&user).Error
    if err != nil {
        return nil, err
    }
    return &user, nil
}

func (r *userRepository) Update(user *model.User) error {
    return r.db.Save(user).Error
}

func (r *userRepository) Delete(id uint) error {
    return r.db.Delete(&model.User{}, id).Error
}

func (r *userRepository) List(offset, limit int) ([]*model.User, error) {
    var users []*model.User
    err := r.db.Offset(offset).Limit(limit).Find(&users).Error
    return users, err
}
```

**internal/service/user_service.go**

```go
package service

import (
    "errors"
    "your-project/internal/model"
    "your-project/internal/repository"
    "gorm.io/gorm"
)

type UserService interface {
    CreateUser(req *model.CreateUserRequest) (*model.User, error)
    GetUser(id uint) (*model.User, error)
    UpdateUser(id uint, req *model.UpdateUserRequest) (*model.User, error)
    DeleteUser(id uint) error
    ListUsers(page, pageSize int) ([]*model.User, error)
}

type userService struct {
    userRepo repository.UserRepository
}

func NewUserService(userRepo repository.UserRepository) UserService {
    return &userService{
        userRepo: userRepo,
    }
}

func (s *userService) CreateUser(req *model.CreateUserRequest) (*model.User, error) {
    // 检查邮箱是否已存在
    if _, err := s.userRepo.GetByEmail(req.Email); err == nil {
        return nil, errors.New("email already exists")
    }
    
    user := &model.User{
        Name:  req.Name,
        Email: req.Email,
    }
    
    if err := s.userRepo.Create(user); err != nil {
        return nil, err
    }
    
    return user, nil
}

func (s *userService) GetUser(id uint) (*model.User, error) {
    return s.userRepo.GetByID(id)
}

func (s *userService) UpdateUser(id uint, req *model.UpdateUserRequest) (*model.User, error) {
    user, err := s.userRepo.GetByID(id)
    if err != nil {
        return nil, err
    }
    
    if req.Name != "" {
        user.Name = req.Name
    }
    if req.Email != "" {
        // 检查新邮箱是否已被其他用户使用
        if existingUser, err := s.userRepo.GetByEmail(req.Email); err == nil && existingUser.ID != id {
            return nil, errors.New("email already exists")
        }
        user.Email = req.Email
    }
    
    if err := s.userRepo.Update(user); err != nil {
        return nil, err
    }
    
    return user, nil
}

func (s *userService) DeleteUser(id uint) error {
    if _, err := s.userRepo.GetByID(id); err != nil {
        if errors.Is(err, gorm.ErrRecordNotFound) {
            return errors.New("user not found")
        }
        return err
    }
    
    return s.userRepo.Delete(id)
}

func (s *userService) ListUsers(page, pageSize int) ([]*model.User, error) {
    offset := (page - 1) * pageSize
    return s.userRepo.List(offset, pageSize)
}
```

**internal/api/handler/user_handler.go**

```go
package handler

import (
    "net/http"
    "strconv"
    "your-project/internal/model"
    "your-project/internal/service"
    
    "github.com/gin-gonic/gin"
)

type UserHandler struct {
    userService service.UserService
}

func NewUserHandler(userService service.UserService) *UserHandler {
    return &UserHandler{
        userService: userService,
    }
}

func (h *UserHandler) CreateUser(c *gin.Context) {
    var req model.CreateUserRequest
    if err := c.ShouldBindJSON(&req); err != nil {
        c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
        return
    }
    
    user, err := h.userService.CreateUser(&req)
    if err != nil {
        c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
        return
    }
    
    c.JSON(http.StatusCreated, gin.H{"data": user})
}

func (h *UserHandler) GetUser(c *gin.Context) {
    idStr := c.Param("id")
    id, err := strconv.ParseUint(idStr, 10, 32)
    if err != nil {
        c.JSON(http.StatusBadRequest, gin.H{"error": "invalid user id"})
        return
    }
    
    user, err := h.userService.GetUser(uint(id))
    if err != nil {
        c.JSON(http.StatusNotFound, gin.H{"error": "user not found"})
        return
    }
    
    c.JSON(http.StatusOK, gin.H{"data": user})
}

func (h *UserHandler) UpdateUser(c *gin.Context) {
    idStr := c.Param("id")
    id, err := strconv.ParseUint(idStr, 10, 32)
    if err != nil {
        c.JSON(http.StatusBadRequest, gin.H{"error": "invalid user id"})
        return
    }
    
    var req model.UpdateUserRequest
    if err := c.ShouldBindJSON(&req); err != nil {
        c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
        return
    }
    
    user, err := h.userService.UpdateUser(uint(id), &req)
    if err != nil {
        c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
        return
    }
    
    c.JSON(http.StatusOK, gin.H{"data": user})
}

func (h *UserHandler) DeleteUser(c *gin.Context) {
    idStr := c.Param("id")
    id, err := strconv.ParseUint(idStr, 10, 32)
    if err != nil {
        c.JSON(http.StatusBadRequest, gin.H{"error": "invalid user id"})
        return
    }
    
    if err := h.userService.DeleteUser(uint(id)); err != nil {
        c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
        return
    }
    
    c.JSON(http.StatusOK, gin.H{"message": "user deleted successfully"})
}

func (h *UserHandler) ListUsers(c *gin.Context) {
    pageStr := c.DefaultQuery("page", "1")
    pageSizeStr := c.DefaultQuery("page_size", "10")
    
    page, _ := strconv.Atoi(pageStr)
    pageSize, _ := strconv.Atoi(pageSizeStr)
    
    if page < 1 {
        page = 1
    }
    if pageSize < 1 || pageSize > 100 {
        pageSize = 10
    }
    
    users, err := h.userService.ListUsers(page, pageSize)
    if err != nil {
        c.JSON(http.StatusInternalServerError, gin.H{"error": err.Error()})
        return
    }
    
    c.JSON(http.StatusOK, gin.H{"data": users})
}
```

```go
package router

import (
    "your-project/internal/api/handler"
    
    "github.com/gin-gonic/gin"
)

func SetupRouter(userHandler *handler.UserHandler) *gin.Engine {
    r := gin.Default()
    
    // 健康检查
    r.GET("/health", func(c *gin.Context) {
        c.JSON(200, gin.H{"status": "ok"})
    })
    
    // API 路由组
    api := r.Group("/api/v1")
    {
        users := api.Group("/users")
        {
            users.POST("", userHandler.CreateUser)
            users.GET("/:id", userHandler.GetUser)
            users.PUT("/:id", userHandler.UpdateUser)
            users.DELETE("/:id", userHandler.DeleteUser)
            users.GET("", userHandler.ListUsers)
        }
    }
    
    return r
}
```

### Wire的工作原理


1. **wire.go 文件**：这是你写的"模板"文件，告诉 Wire 如何组装依赖
2. **wire_gen.go 文件**：这是 Wire 工具自动生成的实际代码，包含真正的依赖注入逻辑
3. **实际运行时**：程序调用的是 `wire_gen.go` 中的代码

#### 具体流程

**第一步：你写 wire.go（模板）**

```go
//go:build wireinject
// +build wireinject

package wire

import (
    "wire-demo/internal/config"
    // ... 其他导入
    "github.com/google/wire"
)

type App struct {
    Engine *gin.Engine  // 注意这里应该是指针
    Config *config.Config
}

func InitializeApp() (*App, error) {
    wire.Build(
        config.LoadConfig,
        database.NewDatabase,
        repository.NewUserRepository,
        service.NewUserService,
        api.NewUserHandler,
        router.SetupRouter,
        wire.Struct(new(App), "*"),
    )
    return &App{}, nil  // 这行代码不会执行，只是占位
}
```

**第二步：运行 wire 命令生成代码**

```bash
cd wire
wire
```

**第三步：Wire 生成 wire_gen.go（实际代码）**

```go
// Code generated by Wire. DO NOT EDIT.

//go:generate go run github.com/google/wire/cmd/wire
//go:build !wireinject
// +build !wireinject

    "wire-demo/internal/api"

    "wire-demo/internal/repository"

    configConfig, err := config.LoadConfig()
    if err != nil {
        return nil, err
    }
    db, err := database.NewDatabase(configConfig)
    if err != nil {
        return nil, err
    }
    userRepository := repository.NewUserRepository(db)
```

#### 关键点说明


1. **构建标签**：
   * `//go:build wireinject` 表示这个文件只在 wire 生成时使用
   * `//go:build !wireinject` 表示这个文件在正常编译时使用
2. **实际调用**：

   ```go
   // main.go 中调用的实际是 wire_gen.go 中的函数
   app, err := wire.InitializeApp()  // 调用生成的代码
   ```
3. **你的 wire.go 有个小问题**：

   ```go
   type App struct {
       Engine *gin.Engine     // 应该是指针
       Config *config.Config  // 应该是指针
   }
   ```

#### 总结

* **wire.go**：你写的模板，告诉 Wire 怎么组装
* **wire_gen.go**：Wire 自动生成的实际代码
* **运行时**：程序执行的是 `wire_gen.go` 中的代码
* **每次修改 wire.go 后都需要重新运行** `wire` 命令

这就是为什么 Wire 被称为"编译时依赖注入"框架，因为依赖关系在编译前就已经确定并生成了具体的代码。

```go
}
```

```go
    var total int64
    
    if err := r.db.Model(&model.User{}).Count(&total).Error; err != nil {
        return nil, 0, err
    }
    
    err := r.db.Offset(offset).Limit(limit).Find(&users).Error
    return users, total, err

    err := r.db.Model(&model.User{}).Where("email = ?", email).Count(&count).Error
```

```go
package service

import (
    "errors"
    "wire-demo/internal/model"
    "wire-demo/internal/repository"
    "gorm.io/gorm"
)

//go:generate mockgen -source=user_service.go -destination=../../mocks/mock_user_service.go

type UserService interface {
    CreateUser(req *model.CreateUserRequest) (*model.User, error)
    GetUser(id uint) (*model.User, error)
    UpdateUser(id uint, req *model.UpdateUserRequest) (*model.User, error)
    DeleteUser(id uint) error
    ListUsers(req *model.ListUsersRequest) (*model.ListUsersResponse, error)
}

type userService struct {
    userRepo repository.UserRepository
}

func NewUserService(userRepo repository.UserRepository) UserService {
    return &userService{
        userRepo: userRepo,
    }
}

func (s *userService) CreateUser(req *model.CreateUserRequest) (*model.User, error) {
    // 检查邮箱是否已存在
    exists, err := s.userRepo.ExistsByEmail(req.Email)
    if err != nil {
        return nil, err
    }
    if exists {
        return nil, errors.New("email already exists")
    }
    
    user := &model.User{
        Name:  req.Name,
        Email: req.Email,
    }
    
    if err := s.userRepo.Create(user); err != nil {
        return nil, err
    }
    
    return user, nil
}

func (s *userService) GetUser(id uint) (*model.User, error) {
    return s.userRepo.GetByID(id)
}

func (s *userService) UpdateUser(id uint, req *model.UpdateUserRequest) (*model.User, error) {
    user, err := s.userRepo.GetByID(id)
    if err != nil {
        if errors.Is(err, gorm.ErrRecordNotFound) {
            return nil, errors.New("user not found")
        }
        return nil, err
    }
    
    if req.Name != "" {
        user.Name = req.Name
    }
    if req.Email != "" {
        // 检查新邮箱是否已被其他用户使用
        if existingUser, err := s.userRepo.GetByEmail(req.Email); err == nil && existingUser.ID != id {
            return nil, errors.New("email already exists")
        }
        user.Email = req.Email
    }
    
    if err := s.userRepo.Update(user); err != nil {
        return nil, err
    }
    
    return user, nil
}

func (s *userService) DeleteUser(id uint) error {
    if _, err := s.userRepo.GetByID(id); err != nil {
        if errors.Is(err, gorm.ErrRecordNotFound) {
            return errors.New("user not found")
        }
        return err
    }
    
    return s.userRepo.Delete(id)
}

func (s *userService) ListUsers(req *model.ListUsersRequest) (*model.ListUsersResponse, error) {
    offset := (req.Page - 1) * req.PageSize
    users, total, err := s.userRepo.List(offset, req.PageSize)
    if err != nil {
        return nil, err
    }
    
    return &model.ListUsersResponse{
        Users: users,
        Total: total,
        Page:  req.Page,
        Size:  req.PageSize,
    }, nil
}
```

```go
package main

import (
    "context"
    "fmt"
    "log"
    "net/http"
    "os"
    "os/signal"
    "syscall"
    "time"
    "wire-demo/wire"
    
    "github.com/gin-gonic/gin"
)

func main() {
    app, err := wire.InitializeApp()
    if err != nil {
        log.Fatal("Failed to initialize app:", err)
    }
    
    // 设置 Gin 运行模式
    gin.SetMode(app.Config.Server.Mode)
    
    // 创建 HTTP 服务器
    addr := fmt.Sprintf(":%s", app.Config.Server.Port)
    srv := &http.Server{
        Addr:         addr,
        Handler:      app.Engine,
        ReadTimeout:  time.Duration(app.Config.Server.ReadTimeout) * time.Second,
        WriteTimeout: time.Duration(app.Config.Server.WriteTimeout) * time.Second,
    }
    
    // 在 goroutine 中启动服务器
    go func() {
        log.Printf("Server starting on %s in %s mode", addr, app.Config.Server.Mode)
        log.Printf("App: %s v%s", app.Config.App.Name, app.Config.App.Version)
        
        if err := srv.ListenAndServe(); err != nil && err != http.ErrServerClosed {
            log.Fatalf("Failed to start server: %v", err)
        }
    }()
    
    // 优雅关闭
    gracefulShutdown(srv)
}

func gracefulShutdown(srv *http.Server) {
    quit := make(chan os.Signal, 1)
    signal.Notify(quit, syscall.SIGINT, syscall.SIGTERM)
    <-quit
    
    log.Println("Shutting down server...")
    
    ctx, cancel := context.WithTimeout(context.Background(), 30*time.Second)
    defer cancel()
    
    if err := srv.Shutdown(ctx); err != nil {
        log.Fatal("Server forced to shutdown:", err)
    }
    
    log.Println("Server exited")
}
```

### wire.Bind 的作用

`wire.Bind` 用于将具体类型绑定到接口类型，当构造函数返回具体类型但你需要接口类型时使用。

**示例对比：**

**不需要 wire.Bind 的情况（推荐）：**

```go
// 构造函数直接返回接口类型
func NewUserRepository(db *gorm.DB) UserRepository {
    return &userRepository{db: db}
}

// Wire 配置
var RepositorySet = wire.NewSet(
    NewUserRepository, // Wire 自动知道这返回 UserRepository 接口
)
```

**需要 wire.Bind 的情况：**

```go
// 构造函数返回具体类型
func NewUserRepository(db *gorm.DB) *userRepository {
    return &userRepository{db: db}
}

// Wire 配置 - 需要手动绑定
var RepositorySet = wire.NewSet(
    NewUserRepository,
    wire.Bind(new(UserRepository), new(*userRepository)), // 告诉 Wire：*userRepository 实现了 UserRepository
)
```

**wire.Bind 的语法：**

```go
wire.Bind(new(接口类型), new(具体类型))
```

### wire.NewSet 的作用

`wire.NewSet` 用于创建一组相关的依赖注入提供者，可以包含：

* 构造函数
* 绑定关系
* 其他 NewSet

**示例：**

```go
// 基础 NewSet
var DatabaseSet = wire.NewSet(
    NewDatabase,
    NewRedis,
)

// 包含绑定的 NewSet
var RepositorySet = wire.NewSet(
    NewUserRepository,
    NewOrderRepository,
    wire.Bind(new(UserRepo), new(*userRepository)),
)

// 组合多个 NewSet
var AppSet = wire.NewSet(
    DatabaseSet,    // 引用其他 NewSet
    RepositorySet,
    ServiceSet,
)
```

**优势：**

* **模块化**：按层或功能组织依赖
* **复用性**：可以在不同的 wire 配置中复用
* **清晰性**：依赖关系更清楚

## Wire 提供了多种方法来处理不同的依赖注入场景。常用的方法：

### 1. 基础方法

#### wire.Build

用于定义依赖注入的构建规则

```go
func InitializeApp() (*App, error) {
    wire.Build(
        NewDatabase,
        NewUserRepository,
        NewUserService,
        NewApp,
    )
    return &App{}, nil
}
```

#### wire.NewSet

创建依赖提供者集合

```go
var DatabaseSet = wire.NewSet(
    NewDatabase,
    NewRedis,
    NewCache,
)
```

### 2. 绑定相关方法

#### wire.Bind

将具体类型绑定到接口

```go
var RepositorySet = wire.NewSet(
    NewUserRepository,
    wire.Bind(new(UserRepository), new(*userRepository)),
)
```

#### wire.BindMap

绑定到 map 类型

```go
// 定义 map 类型
type HandlerMap map[string]Handler

var HandlerSet = wire.NewSet(
    NewUserHandler,
    NewOrderHandler,
    wire.BindMap(new(HandlerMap), "user", new(*UserHandler)),
    wire.BindMap(new(HandlerMap), "order", new(*OrderHandler)),
)
```

### 3. 结构体相关方法

#### wire.Struct

创建结构体，自动注入字段

```go
type App struct {
    UserService  UserService
    OrderService OrderService
    Config       *Config
}

func InitializeApp() (*App, error) {
    wire.Build(
        NewConfig,
        NewUserService,
        NewOrderService,
        wire.Struct(new(App), "*"), // "*" 表示注入所有导出字段
    )
    return &App{}, nil
}

// 也可以指定特定字段
wire.Struct(new(App), "UserService", "Config")
```

#### wire.FieldsOf

从结构体中提取字段作为依赖

```go
type Config struct {
    Database DatabaseConfig
    Redis    RedisConfig
    Server   ServerConfig
}

var ConfigSet = wire.NewSet(
    LoadConfig,
    wire.FieldsOf(new(*Config), "Database", "Redis", "Server"),
)
```

### 4. 值提供方法

#### wire.Value

提供静态值

```go
var AppSet = wire.NewSet(
    wire.Value("localhost:8080"), // 提供字符串值
    wire.Value(30),               // 提供整数值
    NewServer,
)

func NewServer(addr string, timeout int) *Server {
    return &Server{Addr: addr, Timeout: timeout}
}
```

#### wire.InterfaceValue

提供接口值

```go
type Logger interface {
    Log(msg string)
}

type ConsoleLogger struct{}
func (c ConsoleLogger) Log(msg string) { fmt.Println(msg) }

var LoggerSet = wire.NewSet(
    wire.InterfaceValue(new(Logger), ConsoleLogger{}),
)
```


* **Service**: 单元测试，mock Repository
* **Handler**: 单元测试，mock Service
